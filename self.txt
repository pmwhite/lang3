let parse_factor = fun s i len:
  match less_than i len
  | True:
    let c = string_get s i,
    (match c with
     | range 'a' 'z':
       let T i symbol = parse_symbol_assume_first_char s i len,
       (match symbol
        | Name name: T i (Name name)
        | Keyword_fun:
          let i = skip_whitespace s i len,
          let T i args = parse_factor_sequence s i len Empty,
          let i = skip_whitespace s i len,
          let i = skip_exact_char s i len ':',
          let i = skip_whitespace s i len,
          let T i body = parse_expr s i len,
          T i (Fun args body)
        | Keyword_let:
          let i = skip_whitespace s i len,
          let T i pattern = parse_expr s i len,
          let i = skip_whitespace s i len,
          let i = skip_exact_char s i len '=',
          let i = skip_whitespace s i len,
          let T i expr = parse_expr s i len,
          let i = skip_whitespace s i len,
          let i = skip_exact_char s i len ',',
          let i = skip_whitespace s i len,
          let T i body = parse_expr s i len,
          T i (Let pattern expr body)
        | Keyword_match:
          let i = skip_whitespace s i len,
          let T i expr = parse_expr s i len,
          let i = skip_whitespace s i len,
          let T i cases = parse_cases s i len Empty,
          T i (Match expr cases))
     | range 'A' 'Z':
       let buf = buffer_create T,
       buffer_add_char buf c;
       let T i data_name = parse_rest_of_data_name s (int_add i 1) len buf,
       T i (Data data_name)
     | range '0' '9':
       let initial = int_sub (char_code c) (char_code '0'),
       let T i integer = parse_rest_of_integer s (int_add i 1) len initial,
       T i (Integer integer)
    | '"':
      let syntax_buf = buffer_create T,
      let value_buf = buffer_create T,
      let T i syntax value = parse_rest_of_string s (int_add i 1) len '"' syntax_buf value_buf,
      T i (String syntax value)
    | '\'':
      let syntax_buf = buffer_create T,
      let value_buf = buffer_create T,
      let T i syntax value = parse_rest_of_string s (int_add i 1) len '"' syntax_buf value_buf,
      (match string_length value
       | 1: T i (Char syntax (string_get value 0))
       | _: errorfn (Loc s i) "Character literal must only describe single character.")
     | '(':
       let T i expr = parse_expr s (int_add i 1) len,
       let i = skip_exact_char s i len ')',
       T i expr
     | '_': T (int_add i 1) Wildcard
     | _: errorfn (Loc s i) "Expected to find the beginning of an expression, but found an unexpected character '%c' instead." c
    )
  | False: todo,
